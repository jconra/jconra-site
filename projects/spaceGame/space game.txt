var canvas = document.getElementById('myCanvas');
var context = canvas.getContext('2d');

canvas.width = window.innerWidth;
canvas.height = window.innerHeight;

var message="hi";
var angle = 45;
var velocity = 0;
var thrust = 0;

var xcoord = 1.0*(canvas.width/2);     //Always between 200 and 1400 at Fullscreen
var ycoord = 1.0*(canvas.height/2);    //Always between 200 and 800 at Fullscreen
var xinertia = 0;                      //Max at 50
var yinertia = 0;                      //Max at 50
var xmap = 0;
var ymap = 0;
var left = false;
var up = false;
var right = false;
var fire = false;
rocks = new Array();
stars = new Array();

for(i=0;i<50;i++){
rocks[i]=[rn(-100,1800),rn(-100,1100),rn(10,50), Math.floor(i/10)+1, rn(60,100), rn(60,100), rn(60,120), rn(60,120), rn(-60,60), rn(-60,60), rn(-60,60), rn(-60,60), rn(-60,60), rn(-60,60),rn(0,100),rn(0,100)];}
for(i=0;i<500;i++){
stars[i]=[rn(-100,1800),rn(-100,1100),rn(1,3)];}

function rn(min,max){
return Math.floor( Math.random()*(max-min)+min);}

var grd2 = context.createLinearGradient(-20, 0, 20, 0);
	grd2.addColorStop(0,  "#111111"); 
	grd2.addColorStop(.5, "#666666"); 
	grd2.addColorStop(1,  "#111111");
var grd1 = context.createLinearGradient(0, -30, 0, 30);
	grd1.addColorStop(0,  "#111111"); 
	grd1.addColorStop(.5, "#666666"); 
	grd1.addColorStop(1,  "#111111");
var grd3 = context.createLinearGradient(0, 30, 0, 70);
	grd3.addColorStop(0,  "#FFFFFF"); 
	grd3.addColorStop(.2, "#8888FF"); 
	grd3.addColorStop(.4, "#DDDDDD"); 
	grd3.addColorStop(.6, "#5555CC"); 
	grd3.addColorStop(.8, "#999999"); 
	grd3.addColorStop(1,  "#000088");
var grd5 = context.createLinearGradient(0,-10,0,-300);
	grd5.addColorStop(0,  "#FF0000");  
	grd5.addColorStop(1,  "transparent");
   
function draw(){
	context.fillStyle = 'black';
	context.fillRect(0,0,canvas.width,canvas.height);

		var y = ymap+rocks[n][1];
		var s = rocks[n][2]/50;
		context.beginPath();
		context.moveTo(x-rocks[n][6]*s,y+20*s);
		context.bezierCurveTo(x-rocks[n][6]*s, y-20*s, x-rocks[n][5]*s, y-rocks[n][5]*s, x-20*s, y-rocks[n][7]*s);
		context.bezierCurveTo(x+20*s, y-rocks[n][7]*s, x+rocks[n][4]*s, y-rocks[n][4]*s, x+rocks[n][6]*s, y-20*s);
		context.bezierCurveTo(x+rocks[n][6]*s, y+20*s, x+rocks[n][5]*s, y+rocks[n][5]*s, x+20*s, y+rocks[n][7]*s);
		context.bezierCurveTo(x-20*s, y+rocks[n][7]*s, x-rocks[n][4]*s, y+rocks[n][4]*s, x-rocks[n][6]*s, y+20*s);
		var r=(rocks[n][4]+rocks[n][5]+rocks[n][6]+rocks[n][7])/3.5*s;
		var grd = context.createRadialGradient(x,y,1,x,y,r);
		if (rocks[i][3]>2) var blue=40; else var blue=0;
		var color1="rgb("+(blue+rocks[i][3]*12)+","+(blue+rocks[i][3]*8)+","+blue+")";
		var color2="rgb("+Math.floor(rocks[i][3]/2)*12+","+Math.floor(rocks[i][3]/2)*8+",00)";
		grd.addColorStop(0,color1);
		grd.addColorStop(1,color2);
		context.fillStyle=grd;
		context.fill();
		circle(x+rocks[n][8]*s,y+rocks[n][9]*s);
		circle(x+rocks[n][10]*s,y+rocks[n][11]*s);
		circle(x+rocks[n][12]*s,y+rocks[n][13]*s);}
		
	function circle(x,y){
		context.beginPath();
		var grd = context.createRadialGradient(x,y,1,x,y,50);
		grd.addColorStop(0,'rgba(0,0,0,.5)');
		grd.addColorStop(1,'transparent');
		context.fillStyle=grd;
		context.arc(x, y, 50 ,0,Math.PI*2)
		context.fill();}
   
	context.fillStyle = 'white';
	for(i=0;i<100;i++){
		if (xmap-(stars[i][0]*-1)>1800) {stars[i][0]-=1800; stars[i][1]=rn(-100,1100);}
		if (xmap-(stars[i][0]*-1)<-200) {stars[i][0]+=1800; stars[i][1]=rn(-100,1100);}
	    if (ymap-(stars[i][1]*-1)>1100) {stars[i][1]-=1100; stars[i][0]=rn(-100,1800);}
		if (ymap-(stars[i][1]*-1)<-200) {stars[i][1]+=1100; stars[i][0]=rn(-100,1800);}
	context.beginPath();
	var x = xmap+stars[i][0];
	var y = ymap+stars[i][1];
	context.arc(x, y, stars[i][2], 0, Math.PI*2);
	context.fill();}
	


	
	
	for(i=0;i<rocks.length;i++){
	rocks[i][1]-=rocks[i][3];
		if (xmap-(rocks[i][0]*-1)>1800) {rocks[i][0]-=1800; rocks[i][1]=rn(-100,1100);}
		if (xmap-(rocks[i][0]*-1)<-200) {rocks[i][0]+=1800; rocks[i][1]=rn(-100,1100);}
	    if (ymap-(rocks[i][1]*-1)>1100) {rocks[i][1]-=1100; rocks[i][0]=rn(-100,1800);}
		if (ymap-(rocks[i][1]*-1)<-200) {rocks[i][1]+=1100; rocks[i][0]=rn(-100,1800);}
	astroid(i);}

	
	if (fire){
        for (i=0;i<rocks.length;i++){
			if (rocks[i][3]>2){
				var dx=xmap+rocks[i][0]-xcoord;  //horizontal distance
				var dy=ymap+rocks[i][1]-ycoord;  //vertical distance
				var h=Math.sqrt(dx*dx+dy*dy);    //find hypotonise
				if (h<=range){
					var ark=Math.atan(rocks[i][2]/h)*360/Math.PI;
					var ang=Math.atan(dx/(dy*-1))*180/Math.PI;
					if (((dx<0)&&(dy>0))||((dx>0)&&(dy>0)))ang+=180;
					if (angle>270) ang+=360;
					if ((angle>(ang-(ark/2)))&&(angle<(ang+(ark/2)))){
						context.beginPath();
						context.arc(xmap+rocks[i][0], ymap+rocks[i][1], rocks[i][2]*2, 0, Math.PI*2);
						context.fillStyle="white";
						context.fill();
						rocks[i][1]=ymap*-1+900;
						rocks[i][0]=rn(-10,1800);
					}
				}
			}
		}
	}

   
	context.translate(xcoord , ycoord);

	context.lineWidth="2";
	for(i=-1;i<2;i++){
	context.moveTo(30*i,20);

	if (i==0) i++;
	context.moveTo(25*i,-10);
	context.lineTo(0,(range*-1));}
	context.lineWidth="4";
	context.stroke();
	fire=false;}
	context.beginPath();
	context.moveTo(0,-30);
	context.lineTo(40,20);
	context.lineTo(50,50);
	context.lineTo(0,18);
	context.lineTo(-50,50);
	context.lineTo(-40,20);
	context.lineTo(0,-30);
	context.fillStyle = grd1;
	context.fill();
	context.closePath();
	context.beginPath();
	context.moveTo(0,-70);
	context.lineTo(20,30);
	context.lineTo(-20,30);
	context.lineTo(0,-70);
	context.fillStyle = grd2;
	context.fill();
	context.closePath();
	context.beginPath();
	context.moveTo(0,-30);
	context.lineTo(8,-10);
	context.lineTo(-8,-10);

document.onkeydown = KeyDown;
function KeyDown(){
	var KeyID = event.keyCode;
	switch(KeyID){
	case 32:fire=true;             break;       //Spacebar
	case 65:left=true;             break;       //Arrow Left
	case 87:up=true; thrust=5;     break;       //Arrow Up
	case 68:right=true;            break;}}     //Arrow Rigaaaaaaaht
   
document.onkeyup = KeyUp;
function KeyUp(){
	var KeyID = event.keyCode;
	switch(KeyID){
	case 65:left=false;            break;       //Arrow Left
	case 87:up=false;              break;       //Arrow Up
	case 68:right=false;           break;}}     //Arrow Right
   
	function start(){

	   function frame(){
		   if (left) angle-=10;
		   if (up) velocity++;
		   if (right) angle+=10;
		   if (angle>355) angle=0;
	       if (angle<0) angle=355;
		   var radangle=angle/180*Math.PI;
		   var xchange=Math.sin(radangle)*velocity;
		   var ychange=Math.cos(radangle)*velocity;
		   velocity=0;
		   
		   xinertia+=xchange;
		   yinertia+=ychange;
		   
		   if (xinertia>50) xinertia=50;
		   if (yinertia>50) yinertia=50;
		   if (xinertia<-50) xinertia=-50;
		   if (yinertia<-50) yinertia=-50;

		   xcoord+=xinertia;
		   ycoord-=yinertia;
		   
		   if (xcoord<200) {xmap+=(200-xcoord); xcoord=200;}
		   if (xcoord>(canvas.width-200)) {xmap-=xcoord-(canvas.width-200); xcoord=(canvas.width-200);}
		   if (ycoord<200) {ymap+=(200-ycoord); ycoord=200;}
		   if (ycoord>(canvas.height-200)) {ymap-=ycoord-(canvas.height-200); ycoord=(canvas.height-200);}

		   draw();
	   }
		
   var id = setInterval(frame, 50)}

start();
